function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); let data; if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else { throw new Error("No payload received"); } let photoUrl = "No Photo Uploaded"; // Handle base64 photo upload if present if (data.photoUrl && data.photoUrl.startsWith("data:image")) { const folderName = "Maintenance Ticket Photos"; let folder; const folders = DriveApp.getFoldersByName(folderName); if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder(folderName); } const contentType = data.photoUrl.substring(5, data.photoUrl.indexOf(";")); const base64Data = data.photoUrl.substring(data.photoUrl.indexOf(",") + 1); const bytes = Utilities.base64Decode(base64Data); const blob = Utilities.newBlob(bytes, contentType, "photo_" + new Date().getTime() + ".jpg"); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); photoUrl = file.getUrl(); } // Append submission row to Google Sheet sheet.appendRow([ new Date(), // Timestamp data.teacherName, // Teacher Name data.classroom, // Classroom / Location data.issue, // Issue Description data.urgency, // Urgency photoUrl, // Photo Link "Pending", // Status "", // Assigned To "" // Notes / Resolution ]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
